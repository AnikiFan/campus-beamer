\begin{frame}{只提供思路与素材，交给 agent 完成}
      \begin{description}[agent 完成]
            \item[你的思路] 用一段话或大纲说明想讲什么、面向谁、重点是什么。
            \item[本地素材] 在 \texttt{materials/} 准备论文 LaTeX 项目、PDF、图片或实验结果。
            \item[agent 完成] 阅读材料、组织内容、选择图表、核对引用，检查排版并编写讲解备注。
            \item[最终交付] \texttt{build/main.pdf} 和带逐页备注的 \texttt{build/main.pptx}。
      \end{description}
      \medskip
      提示词参考 \texttt{prompt.md}；后续直接告诉 agent 需要调整的内容。
\end{frame}
