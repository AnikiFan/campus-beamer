\subsection{样式演示}

\begin{frame}{提示、备注与警告}
      \begin{tipbox}[提示]
            使用 \texttt{tipbox} 放置操作建议，例如推荐的实验配置。
      \end{tipbox}

      \begin{notebox}[备注]
            使用 \texttt{notebox} 放置补充说明，例如数据来源或实验条件。
      \end{notebox}

      \begin{alertbox}[警告]
            使用 \texttt{alertbox} 强调易错点，例如复现时需要核对的设置。
      \end{alertbox}
\end{frame}
\note{比较提示、备注与警告的功能色。每个框只放一个需要强调的信息。}

\subsection{使用方法}

\begin{frame}[fragile]{提示框用法}
      \begin{verbatim}
\begin{tipbox}[自定义标题]
  需要强调的内容。
\end{tipbox}
      \end{verbatim}
      \begin{itemize}
            \item 将环境名替换为 \texttt{notebox}、\texttt{alertbox}、\texttt{examplebox} 或 \texttt{definitionbox}。
            \item 省略标题使用默认名称；无标题框用 \texttt{campusbox}。
      \end{itemize}
      \begin{tipbox}
            省略可选标题时，会使用默认标题。
      \end{tipbox}
\end{frame}
\note{展示提示框环境的写法；省略标题会使用当前主题语言的默认名称。campusbox 完全去掉标题栏，其可选参数为颜色，默认 tipcolor。}


\begin{frame}[fragile]{无标题正文框}
  \begin{verbatim}
\begin{campusbox}
  一次只强调一个重点。
\end{campusbox}
  \end{verbatim}
  \begin{campusbox}
    一次只强调一个重点。
  \end{campusbox}
  \begin{campusbox}[notecolor]
    可选参数选择配色，例如 \texttt{[notecolor]}。
  \end{campusbox}
  \begin{campusbox}[alertcolor]
    标题栏及其空间均被移除，正文保留左侧色条。
  \end{campusbox}
\end{frame}
\note{本页三个框都没有标题。campusbox 默认采用 tipcolor，可选参数为颜色而非标题。上方代码对应第一个框；后两个框展示备注和警告配色。}
